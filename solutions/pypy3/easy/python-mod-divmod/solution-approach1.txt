# Enter your code here. Read input from STDIN. Print output to STDOUT

a = int(input())
b = int(input())

# 1. First line: Integer division (quotient)
print(a // b)

# 2. Second line: Modulo operator (remainder)
print(a % b)

# 3. Third line: The divmod tuple
print(divmod(a, b))
